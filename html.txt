<script>
  const API_URL = 'https://script.google.com/macros/s/AKfycbxAzKZ_YztFop15snUuZxjYt10LTSmM_g4LLHePYYZGCes7K9LBg-kn0Cmg2BBSgDJR/exec';
  const PINK = '#ec6a9b';

  const $q = document.getElementById('q');
  const $btn = document.getElementById('btnSearch');
  const $clear = document.getElementById('btnClear');
  const $result = document.getElementById('result');
  const $count = document.getElementById('count');
  const btnHTML = $btn.innerHTML;

  const esc = s => String(s ?? '').replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));

  const skeleton = () => {
    $count.textContent = '';
    $result.innerHTML = '<div class="skel"></div><div class="skel" style="height:150px"></div>';
  };

  async function search() {
    const q = $q.value.trim();
    if (q.length < 2) {
      Swal.fire({ icon:'warning', title:'กรุณากรอกชื่อหรือนามสกุล', text:'พิมพ์อย่างน้อย 2 ตัวอักษร', confirmButtonColor: PINK });
      return;
    }
    $btn.disabled = true;
    $btn.innerHTML = '<span class="spinner-border spinner-border-sm"></span>';
    skeleton();

    try {
      const res = await fetch(API_URL + '?q=' + encodeURIComponent(q));
      const data = await res.json();
      render(data.results || []);
    } catch (err) {
      console.error(err);
      $result.innerHTML = '';
      Swal.fire({ icon:'error', title:'เกิดข้อผิดพลาด', text:'ไม่สามารถเชื่อมต่อระบบได้ กรุณาลองใหม่อีกครั้ง', confirmButtonColor: PINK });
    } finally {
      $btn.disabled = false;
      $btn.innerHTML = btnHTML;
    }
  }

  function render(list) {
    if (!list.length) {
      $count.textContent = '';
      $result.innerHTML = `
        <div class="glass empty">
          <div class="emoji">🔍</div>
          <p class="mb-0 mt-2 fw-semibold" style="color:var(--pink-600)">ไม่พบข้อมูลของคุณในระบบ</p>
          <p class="hint mb-0 mt-1">ลองตรวจสอบตัวสะกด หรือค้นหาด้วยนามสกุลแทน</p>
        </div>`;
      return;
    }

    $count.innerHTML = `พบ <b>${list.length}</b> รายการ`;
    $result.innerHTML = list.map((r, i) => {
      const name = `${r.prename || ''}${r.firstname || ''} ${r.lastname || ''}`.trim();
      const initial = esc((r.firstname || r.lastname || '?').trim().charAt(0));
      const action = r.viewUrl
        ? `<a href="${esc(r.viewUrl)}" target="_blank" rel="noopener" data-name="${esc(name)}"
              class="btn btn-pink btn-download w-100 mt-3">
              <svg width="20" height="20" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.4" stroke-linecap="round" stroke-linejoin="round"><path d="M12 3v12m0 0 4.5-4.5M12 15l-4.5-4.5"/><path d="M4 17v2a2 2 0 0 0 2 2h12a2 2 0 0 0 2-2v-2"/></svg>
              ดาวน์โหลด PDF</a>`
        : `<button class="btn btn-light w-100 mt-3 py-3 rounded-pill" disabled>ยังไม่มีไฟล์ PDF</button>`;

      return `
        <article class="result-card" style="animation-delay:${i * 70}ms">
          <div class="d-flex align-items-center gap-3 mb-3">
            <div class="avatar">${initial}</div>
            <div class="min-w-0">
              <div class="rname">${esc(name)}</div>
              <span class="chip mt-1">🏅 รหัส ${esc(r.id)}</span>
            </div>
          </div>
          <div class="rtitle">
            <small>ชื่อเรื่อง</small>
            <span>${esc(r.title) || '-'}</span>
          </div>
          ${action}
        </article>`;
    }).join('');
  }

  // แจ้งเตือนเมื่อกดดาวน์โหลด (ลิงก์เปิดในแท็บใหม่ตามปกติ)
  $result.addEventListener('click', e => {
    const a = e.target.closest('.btn-download');
    if (!a) return;
    Swal.fire({
      icon:'success', title:'ดาวน์โหลดสำเร็จ',
      text:'เปิดไฟล์เกียรติบัตรของ ' + a.dataset.name + ' แล้ว',
      timer: 2500, timerProgressBar: true, showConfirmButton:false
    });
  });

  $q.addEventListener('input', () => $clear.classList.toggle('show', $q.value.length > 0));
  $clear.addEventListener('click', () => {
    $q.value = ''; $clear.classList.remove('show');
    $result.innerHTML = ''; $count.textContent = ''; $q.focus();
  });
  $btn.addEventListener('click', search);
  $q.addEventListener('keydown', e => { if (e.key === 'Enter') search(); });
</script>
